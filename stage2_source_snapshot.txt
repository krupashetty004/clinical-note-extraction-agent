"""Strict schemas and original-note evidence validation."""

import re
from typing import Annotated, Literal

from pydantic import (
    BaseModel,
    ConfigDict,
    Field,
    StrictInt,
    StrictStr,
    model_validator,
)


EntityCategory = Literal[
    "diagnosis",
    "medication",
    "procedure",
    "allergy",
    "vital_sign",
]

DiagnosisStatus = Literal[
    "active",
    "historical",
    "ruled_out",
    "suspected",
]

MedicationStatus = Literal[
    "current",
    "discontinued",
    "newly_prescribed",
]

MentionStatus = DiagnosisStatus | MedicationStatus


class ExtractedItem(BaseModel):
    """One clinical mention with evidence in the original source note."""

    model_config = ConfigDict(extra="forbid")

    category: EntityCategory
    entity: StrictStr = Field(min_length=1)

    # Diagnoses and medications have their own status vocabularies.
    # Procedures, allergies, and vital signs have no status field
    # under the current SOW contract.
    status: MentionStatus | None = None

    evidence_text: StrictStr = Field(min_length=1)

    # Zero-based, end-exclusive offsets into the original Python string.
    start_offset: Annotated[StrictInt, Field(ge=0)]
    end_offset: Annotated[StrictInt, Field(gt=0)]

    # Values explicitly stated in the source, such as medication dose,
    # route/frequency, procedure date, allergy reaction, or vital/unit.
    attributes: dict[str, StrictStr] = Field(default_factory=dict)

    @model_validator(mode="after")
    def validate_item_fields(self) -> "ExtractedItem":
        """Enforce status rules according to the entity category."""

        if not self.entity.strip():
            raise ValueError(
                "Entity must not be empty or whitespace-only."
            )

        if not self.evidence_text.strip():
            raise ValueError(
                "Evidence text must not be empty or whitespace-only."
            )

        if self.end_offset <= self.start_offset:
            raise ValueError(
                "end_offset must be greater than start_offset."
            )

        if self.category == "diagnosis":
            if self.status not in {
                "active",
                "historical",
                "ruled_out",
                "suspected",
            }:
                raise ValueError(
                    "Diagnosis status must be active, historical, "
                    "ruled_out, or suspected."
                )

        elif self.category == "medication":
            if self.status not in {
                "current",
                "discontinued",
                "newly_prescribed",
            }:
                raise ValueError(
                    "Medication status must be current, "
                    "discontinued, or newly_prescribed."
                )

        elif self.status is not None:
            raise ValueError(
                "Status is not defined for this entity category "
                "in the current extraction contract."
            )

        return self


class ExtractionResult(BaseModel):
    """Structured result returned by the extraction component."""

    model_config = ConfigDict(extra="forbid")

    items: list[ExtractedItem]
    needs_review: bool
    review_reasons: list[StrictStr]


class EvidenceSpanValidationError(ValueError):
    """Typed evidence-span failure, without exposing source text."""

    def __init__(self, code: str, message: str) -> None:
        super().__init__(message)
        self.code = code


_WORD_PATTERN = re.compile(r"[^\W_]+", re.UNICODE)
_NEGATION_CUES = (
    "no ",
    "no evidence of",
    "denies ",
    "denied ",
    "without ",
    "negative for",
    "not ",
    "never ",
    "ruled out",
    "has not been diagnosed",
)
_HISTORICAL_CUES = (
    "history of",
    "historical",
    "previously",
    "formerly",
    "in the past",
    "remote ",
)
_UNCERTAINTY_CUES = (
    "possible ",
    "possibly ",
    "suspected ",
    "likely ",
    "may have",
    "might have",
    "question of",
    "concern for",
)
_CURRENT_CUES = (
    "current ",
    "currently ",
    "active ",
    "present ",
)
_DISCONTINUED_CUES = (
    "discontinued",
    "stopped",
    "held",
    "no longer taking",
)
_NEW_MEDICATION_CUES = (
    "new prescription",
    "newly prescribed",
    "prescribed",
    "started",
    "begin ",
    "began ",
)


def _normalized_tokens(value: str) -> set[str]:
    return set(_WORD_PATTERN.findall(value.casefold()))


def entity_evidence_is_grounded(entity: str, evidence_text: str) -> bool:
    """Return whether every entity token occurs in cited evidence."""
    entity_tokens = _normalized_tokens(entity)
    evidence_tokens = _normalized_tokens(evidence_text)
    return bool(entity_tokens) and entity_tokens <= evidence_tokens


def _context_for_span(note: str, start: int, end: int) -> str:
    """Return the sentence or line containing a source span."""
    left_boundary = max(
        note.rfind(".", 0, start),
        note.rfind("!", 0, start),
        note.rfind("?", 0, start),
        note.rfind(";", 0, start),
        note.rfind("\n", 0, start),
    )
    right_candidates = [
        position
        for position in (
            note.find(".", end),
            note.find("!", end),
            note.find("?", end),
            note.find(";", end),
            note.find("\n", end),
        )
        if position >= 0
    ]
    right_boundary = min(right_candidates, default=len(note))
    return note[left_boundary + 1:right_boundary + 1].casefold()


def _contains_cue(context: str, cues: tuple[str, ...]) -> bool:
    return any(cue in context for cue in cues)


def _validate_item_grounding(
    item: ExtractedItem,
    original_note: str,
) -> None:
    """Reject fabricated entities, attributes, or status context."""
    if not entity_evidence_is_grounded(
        item.entity,
        item.evidence_text,
    ):
        raise EvidenceSpanValidationError(
            "evidence_entity_mismatch",
            "The evidence does not contain the extracted entity.",
        )

    for attribute_name, attribute_value in item.attributes.items():
        if not attribute_name.strip() or not attribute_value.strip():
            raise EvidenceSpanValidationError(
                "evidence_attribute_invalid",
                "Evidence attributes must have non-blank names and values.",
            )
        if attribute_value.casefold() not in item.evidence_text.casefold():
            raise EvidenceSpanValidationError(
                "evidence_attribute_mismatch",
                "An extracted attribute is not present in the evidence.",
            )

def validate_item_status_context(
    item: ExtractedItem,
    original_note: str,
) -> str | None:
    """Return a status-context error without rejecting extraction output.

    Status reconciliation belongs to the independent verifier. Extraction
    must retain a source-grounded mention even when its predicted status is
    wrong, so the verifier can flag it for review.
    """
    context = _context_for_span(
        original_note,
        item.start_offset,
        item.end_offset,
    )
    if item.category == "diagnosis":
        if item.status == "active" and (
            _contains_cue(context, _NEGATION_CUES)
            or _contains_cue(context, _HISTORICAL_CUES)
            or _contains_cue(context, _UNCERTAINTY_CUES)
        ):
            return (
                "Active diagnosis evidence conflicts with its source context."
            )
        if item.status == "historical":
            if _contains_cue(context, _CURRENT_CUES):
                return (
                    "Historical diagnosis status conflicts with current "
                    "source context."
                )
            if not _contains_cue(context, _HISTORICAL_CUES):
                return (
                    "Historical diagnosis status lacks historical source "
                    "context."
                )
        if item.status == "suspected" and not _contains_cue(
            context, _UNCERTAINTY_CUES
        ):
            return (
                "Suspected diagnosis status lacks uncertainty source context."
            )
        if item.status == "ruled_out" and not _contains_cue(
            context, _NEGATION_CUES
        ):
            return (
                "Ruled-out diagnosis status lacks negating source context."
            )
    elif item.category == "medication":
        if item.status == "current" and (
            _contains_cue(context, _DISCONTINUED_CUES)
            or _contains_cue(context, _HISTORICAL_CUES)
            or _contains_cue(context, _UNCERTAINTY_CUES)
        ):
            return "Current medication status conflicts with source context."
        if item.status == "discontinued" and not _contains_cue(
            context, _DISCONTINUED_CUES
        ):
            return "Discontinued medication status lacks source context."
        if item.status == "newly_prescribed" and not _contains_cue(
            context, _NEW_MEDICATION_CUES
        ):
            return "New medication status lacks prescription source context."
    return None


def validate_evidence_spans(
    result: ExtractionResult,
    original_note: str,
) -> None:
    """Validate exact evidence text and offsets against the original note.

    Raises:
        TypeError: If original_note is not a string.
        ValueError: If evidence is outside the note or differs from
            the original substring.

    An exact substring match does not prove semantic support.
    """
    if not isinstance(original_note, str):
        raise TypeError("original_note must be a string.")

    for index, item in enumerate(result.items):
        if item.end_offset > len(original_note):
            raise EvidenceSpanValidationError(
                "evidence_offset_out_of_bounds",
                f"Item {index}: end_offset exceeds the original note length.",
            )

        actual_evidence = original_note[
            item.start_offset:item.end_offset
        ]

        if actual_evidence != item.evidence_text:
            raise EvidenceSpanValidationError(
                "evidence_text_offset_mismatch",
                f"Item {index}: evidence_text does not match "
                "the original note at the supplied offsets.",
            )

        _validate_item_grounding(item, original_note)

class LocalClinicalExtractionProvider:
    """Extract clinical entities strictly grounded in the original source note."""

    def extract(self, input_text: str) -> dict[str, Any]:
        if not isinstance(input_text, str):
            raise TypeError("input_text must be a string.")

        text_lower = input_text.lower()
        items: list[dict[str, Any]] = []
        occupied_spans: list[tuple[int, int]] = []

        def overlaps(s: int, e: int) -> bool:
            return any(max(s, os) < min(e, oe) for os, oe in occupied_spans)

        category_map = {
            "diagnoses": "diagnosis",
            "medications": "medication",
            "procedures": "procedure",
            "allergies": "allergy",
            "vital_signs": "vital_sign",
        }

        for group_key, terms in CLINICAL_TERMS.items():
            category = category_map[group_key]
            for term in sorted(terms, key=len, reverse=True):
                cursor = 0
                while True:
                    idx = text_lower.find(term, cursor)
                    if idx == -1:
                        break
                    end = idx + len(term)
                    cursor = end

                    prev_char = text_lower[idx - 1] if idx > 0 else " "
                    next_char = text_lower[end] if end < len(text_lower) else " "
                    if prev_char.isalnum() or next_char.isalnum():
                        continue

                    if overlaps(idx, end):
                        continue

                    evidence = input_text[idx:end]
                    status = None
                    if category == "diagnosis":
                        window = text_lower[max(0, idx - 40):idx]
                        if any(w in window for w in ["history of", "hx of", "past history", "prior"]):
                            status = "historical"
                        elif any(w in window for w in ["no evidence of", "ruled out", "negative for", "denies"]):
                            status = "ruled_out"
                        elif any(w in window for w in ["possible", "probable", "rule out", "suspected", "question of"]):
                            status = "suspected"
                        else:
                            status = "active"

                    elif category == "medication":
                        window = text_lower[max(0, idx - 40):idx]
                        if any(w in window for w in ["discontinued", "stopped", "off", "d/c"]):
                            status = "discontinued"
                        elif any(w in window for w in ["started on", "prescribed", "begin", "newly"]):
                            status = "newly_prescribed"
                        else:
                            status = "current"

                    items.append({
                        "category": category,
                        "entity": evidence,
                        "evidence_text": evidence,
                        "start_offset": idx,
                        "end_offset": end,
                        "status": status,
                        "attributes": {},
                    })
                    occupied_spans.append((idx, end))

        # Sort by start offset
        items.sort(key=lambda x: x["start_offset"])

        return {
            "items": items,
            "needs_review": False,
            "review_reasons": [],
        }


class LocalClinicalVerificationProvider:
    """Verify evidence integrity, offsets, contradictions, and status.

    Returns a payload conforming to the VerificationResult schema expected
    by the orchestrator's validate_verification_output() contract:
    each item_check must include item_index, outcome, status_check, reason,
    and evidence fields.
    """

    # Categories that carry a status field in the extraction contract.
    _STATUS_CATEGORIES = {"diagnosis", "medication"}

    def verify(
        self,
        original_note: str,
        extraction: ExtractionResult,
        coding: DiagnosisCodingBatch,
    ) -> dict[str, Any]:
        if not isinstance(original_note, str):
            raise TypeError("original_note must be a string.")

        item_checks: list[dict[str, Any]] = []
        needs_review = False
        review_reasons: list[str] = []

        for index, item in enumerate(extraction.items):
            start = item.start_offset
            end = item.end_offset
            evidence = item.evidence_text
            status_applies = item.category in self._STATUS_CATEGORIES

            # Check offset and evidence validity
            if (
                start < 0
                or end > len(original_note)
                or end <= start
                or original_note[start:end] != evidence
            ):
                item_checks.append({
                    "item_index": index,
                    "outcome": "unsupported",
                    "status_check": "correct" if status_applies else "not_applicable",
                    "reason": "Evidence text does not match original note at specified character offsets.",
                    "evidence": [],
                })
                needs_review = True
                review_reasons.append(f"Item #{index} offset mismatch.")
                continue

            # Check grounding of entity tokens in evidence
            if not _tokens(item.entity).issubset(_tokens(evidence)):
                item_checks.append({
                    "item_index": index,
                    "outcome": "unsupported",
                    "status_check": "correct" if status_applies else "not_applicable",
                    "reason": "Entity is not grounded in the quoted evidence text.",
                    "evidence": [],
                })
                needs_review = True
                review_reasons.append(f"Item #{index} entity ungrounded in evidence.")
                continue

            # Build evidence reference for supported items
            evidence_ref = {
                "evidence_text": evidence,
                "start_offset": start,
                "end_offset": end,
            }

            # Determine status_check for diagnoses and medications
            if status_applies:
                status_check = "correct"
                # Check for contextual contradictions in a short window before the span
                window = original_note[max(0, start - 60): start].lower()
                if item.category == "diagnosis":
                    if item.status == "active" and any(
                        w in window
                        for w in ["no evidence of", "ruled out", "negative for", "denies", "not"]
                    ):
                        status_check = "uncertain"
                        review_reasons.append(f"Item #{index} active status may conflict with negation context.")
                    elif item.status == "historical" and any(
                        w in window for w in ["history of", "hx of", "past", "prior"]
                    ):
                        status_check = "correct"
                    elif item.status == "ruled_out" and not any(
                        w in window
                        for w in ["no evidence of", "ruled out", "negative for", "denies", "not"]
                    ):
                        status_check = "uncertain"
                    elif item.status == "suspected" and not any(
                        w in window
                        for w in ["possible", "probable", "suspected", "question of", "rule out"]
                    ):
                        status_check = "uncertain"
                elif item.category == "medication":
                    if item.status == "discontinued" and not any(
                        w in window for w in ["discontinued", "stopped", "off", "d/c"]
                    ):
                        status_check = "uncertain"
            else:
                status_check = "not_applicable"

            item_checks.append({
                "item_index": index,
                "outcome": "supported",
                "status_check": status_check,
                "reason": "Evidence directly supports the extracted entity in the original note.",
                "evidence": [evidence_ref],
            })

            if status_check in {"incorrect", "uncertain"}:
                needs_review = True

        if needs_review and not review_reasons:
            review_reasons.append("One or more items require clinical review.")

        return {
            "item_checks": item_checks,
            "potential_omissions": [],
            "needs_review": needs_review,
            "review_reasons": review_reasons,
        }


===== .\dashboard\extraction\verification.py =====
"""Validation contract for independent clinical extraction verification.

This module validates a verifier's structured output. It does not call an
AI provider or determine clinical truth on its own.
"""

from typing import Annotated, Literal

from pydantic import (
    BaseModel,
    ConfigDict,
    Field,
    StrictBool,
    StrictInt,
    StrictStr,
    ValidationError,
    model_validator,
)

from dashboard.extraction.schema import (
    EntityCategory,
    ExtractionResult,
    entity_evidence_is_grounded,
    validate_item_status_context,
    validate_evidence_spans,
)


VerificationOutcome = Literal[
    "supported",
    "unsupported",
    "contradicted",
    "uncertain",
]

StatusCheck = Literal[
    "correct",
    "incorrect",
    "uncertain",
    "not_applicable",
]


class EvidenceReference(BaseModel):
    """A quoted source span used by the verifier."""

    model_config = ConfigDict(extra="forbid", frozen=True)

    evidence_text: StrictStr = Field(min_length=1)
    start_offset: Annotated[StrictInt, Field(ge=0)]
    end_offset: Annotated[StrictInt, Field(gt=0)]

    @model_validator(mode="after")
    def validate_span(self) -> "EvidenceReference":
        if not self.evidence_text.strip():
            raise ValueError("Evidence text must not be blank.")

        if self.end_offset <= self.start_offset:
            raise ValueError(
                "end_offset must be greater than start_offset."
            )

        return self


class ItemVerification(BaseModel):
    """Verification findings for one extracted item."""

    model_config = ConfigDict(extra="forbid", frozen=True)

    item_index: Annotated[StrictInt, Field(ge=0)]
    outcome: VerificationOutcome
    status_check: StatusCheck
    reason: StrictStr = Field(min_length=1)
    evidence: tuple[EvidenceReference, ...] = ()

    @model_validator(mode="after")
    def validate_reason(self) -> "ItemVerification":
        if not self.reason.strip():
            raise ValueError("Verification reason must not be blank.")

        return self


class PotentialOmission(BaseModel):
    """A possible clinical mention missing from the extraction."""

    model_config = ConfigDict(extra="forbid", frozen=True)

    category: EntityCategory
    entity: StrictStr = Field(min_length=1)
    evidence: EvidenceReference
    reason: StrictStr = Field(min_length=1)

    @model_validator(mode="after")
    def validate_text(self) -> "PotentialOmission":
        if not self.entity.strip():
            raise ValueError("Omission entity must not be blank.")

        if not self.reason.strip():
            raise ValueError("Omission reason must not be blank.")

        return self


class VerificationResult(BaseModel):
    """Structured response expected from a future verification pass."""

    model_config = ConfigDict(extra="forbid", frozen=True)

    item_checks: tuple[ItemVerification, ...] = ()
    potential_omissions: tuple[PotentialOmission, ...] = ()
    needs_review: StrictBool
    review_reasons: tuple[StrictStr, ...] = ()

    @model_validator(mode="after")
    def validate_review_decision(self) -> "VerificationResult":
        """Prevent flagged findings from being silently accepted."""

        if any(not reason.strip() for reason in self.review_reasons):
            raise ValueError("Review reasons must not be blank.")

        unresolved_item = any(
            check.outcome != "supported"
            or check.status_check in {"incorrect", "uncertain"}
            for check in self.item_checks
        )

        review_required = (
            unresolved_item or bool(self.potential_omissions)
        )

        if self.needs_review and not self.review_reasons:
            raise ValueError(
                "A result needing review must include review reasons."
            )

        if not self.needs_review and (
            review_required or self.review_reasons
        ):
            raise ValueError(
                "Unresolved findings must be marked for human review."
            )

        return self


class VerificationOutputError(ValueError):
    """Raised when a verifier response fails contract validation."""

    def __init__(
        self,
        code: str,
        message: str,
        *,
        detail_code: str | None = None,
    ) -> None:
        super().__init__(message)
        self.code = code
        self.detail_code = detail_code


def validate_verification_output(
    payload: object,
    extraction: ExtractionResult,
    original_note: str,
) -> VerificationResult:
    """Validate verifier output against an extraction and its source note.

    Checks:
    - The original extraction's evidence spans are valid.
    - Every extracted item is checked exactly once.
    - Status checks match the entity category.
    - Evidence references match the exact original-note substring.
    - Unresolved findings cannot be marked as requiring no review.

    This validates structure and source references, not semantic correctness.
    """
    if not isinstance(original_note, str):
        raise TypeError("original_note must be a string.")

    if not isinstance(extraction, ExtractionResult):
        raise TypeError("extraction must be an ExtractionResult.")

    try:
        validate_evidence_spans(extraction, original_note)
    except (ValueError, TypeError) as exc:
        raise VerificationOutputError(
            "input_extraction_invalid",
            "The input extraction has invalid source evidence.",
        ) from exc

    try:
        result = VerificationResult.model_validate(payload)
    except ValidationError as exc:
        raise VerificationOutputError(
            "verification_schema_failed",
            "Verification output failed schema validation.",
        ) from exc

    # Every extracted item must be checked exactly once.
    indices = [check.item_index for check in result.item_checks]
    expected_indices = set(range(len(extraction.items)))

    if (
        len(indices) != len(set(indices))
        or set(indices) != expected_indices
    ):
        raise VerificationOutputError(
            "verification_coverage_failed",
            "Verification must check every extracted item exactly once.",
        )

    # Only diagnoses and medications have status fields in the contract.
    status_categories = {"diagnosis", "medication"}

    for check in result.item_checks:
        item = extraction.items[check.item_index]
        status_applies = item.category in status_categories

        if status_applies and check.status_check == "not_applicable":
            raise VerificationOutputError(
                "verification_status_check_invalid",
                "Diagnosis and medication status must be checked.",
            )

        if not status_applies and check.status_check != "not_applicable":
            raise VerificationOutputError(
                "verification_status_check_invalid",
                "Status checks are not applicable to this category.",
            )

        status_context_error = validate_item_status_context(
            item,
            original_note,
        )
        if status_context_error is not None and check.status_check == "correct":
            raise VerificationOutputError(
                "verification_status_check_invalid",
                "Verification marked a source status conflict as correct.",
                detail_code="status_context_mismatch",
            )

        if (
            check.outcome in {"supported", "contradicted"}
            and not check.evidence
        ):
            raise VerificationOutputError(
                "verification_evidence_failed",
                "Supported or contradicted findings require evidence.",
                detail_code="required_evidence_missing",
            )
        if check.evidence and not entity_evidence_is_grounded(
            item.entity,
            " ".join(
                reference.evidence_text
                for reference in check.evidence
            ),
        ):
            raise VerificationOutputError(
                "verification_evidence_failed",
                "Verification evidence does not support the extracted entity.",
                detail_code="evidence_entity_mismatch",
            )

    references = [
        reference
        for check in result.item_checks
        for reference in check.evidence
    ]
    references.extend(
        omission.evidence
        for omission in result.potential_omissions
    )

    for reference in references:
        if reference.end_offset > len(original_note):
            raise VerificationOutputError(
                "verification_evidence_failed",
                "A verification evidence span exceeds the original note.",
                detail_code="evidence_offset_out_of_bounds",
            )

        actual_text = original_note[
            reference.start_offset:reference.end_offset
        ]

        if actual_text != reference.evidence_text:
            raise VerificationOutputError(
                "verification_evidence_failed",
                "A verification evidence span does not match the original note.",
                detail_code="evidence_text_offset_mismatch",
            )

    for omission in result.potential_omissions:
        if not entity_evidence_is_grounded(
            omission.entity,
            omission.evidence.evidence_text,
        ):
            raise VerificationOutputError(
                "verification_evidence_failed",
                "Omission evidence does not support the omitted entity.",
                detail_code="evidence_entity_mismatch",
            )

    return result

# ---------------------------------------------------------------------------

def _make_valid_item(note: str, term: str, category: str, status: str) -> ExtractedItem:
    """Build an ExtractedItem with exact offsets grounded in note."""
    idx = note.lower().find(term.lower())
    if idx == -1:
        raise ValueError(f"{term!r} not found in note.")
    end = idx + len(term)
    return ExtractedItem(
        category=category,
        entity=note[idx:end],
        evidence_text=note[idx:end],
        start_offset=idx,
        end_offset=end,
        status=status,
        attributes={},
    )


def _make_extraction_result(items: list[ExtractedItem]) -> ExtractionResult:
    """Wrap a list of ExtractedItem objects into an ExtractionResult."""
    return ExtractionResult(items=items, needs_review=False, review_reasons=[])


def _make_invalid_extraction(items_data: list[dict]) -> ExtractionResult:
    """Build an ExtractionResult bypassing Pydantic validation (for failure tests)."""
    parsed_items = [
        ExtractedItem.model_construct(**item) for item in items_data
    ]
    return ExtractionResult.model_construct(
        items=parsed_items, needs_review=False, review_reasons=[]
    )


def _empty_coding() -> DiagnosisCodingBatch:
    return DiagnosisCodingBatch()

class SyntheticExtractionProvider:
    """Deterministic extraction provider for tests."""

    def __init__(self, responses):
        self.responses = list(responses)
        self.calls = 0

    def extract(self, input_text):
        self.calls += 1
        response = self.responses.pop(0)

        if isinstance(response, Exception):
            raise response

        return response


class SyntheticVerificationProvider:
    """Deterministic verification provider for tests."""

    def __init__(self, responses):
        self.responses = list(responses)
        self.calls = 0

    def verify(self, original_note, extraction, coding):
        self.calls += 1
        response = self.responses.pop(0)

        if isinstance(response, Exception):
            raise response

        return response
